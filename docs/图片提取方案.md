要解析微信公众号、小红书和抖音图文的链接并提取其中的图片，核心难点在于**突破平台的反爬机制（如 Token 校验、动态签名）**以及**处理防盗链（导致图片在非官方域名下显示裂图）**。

因为小红书和抖音的网页版有严格的加密签名（如小红书的 `X-Sign`），纯前端 Vue3 直接请求不仅会触发跨域（CORS），也无法计算签名。因此，**必须采用 Node.js 作为后端/代理层**来完成解析。

以下为你梳理一套完整的 **Vue3 + Node.js（Express/Koa）** 实现方案。

---

## 方案核心思路

1. **公众号**：相对最简单。文章内容是直出的 HTML，通过 Node.js 请求页面，用 `cheerio` 解析 HTML，捞取 `<img>` 标签的 `data-src` 属性即可。
2. **小红书/抖音图文**：页面是 SSR（服务端渲染）或通过 API 异步加载。
* **轻量快捷方案**：Node.js 抓取网页源码，利用正则或 `cheerio` 寻找嵌入在 HTML 中的隐藏 JSON 状态数据（例如小红书的 `window.__INITIAL_STATE__`），直接从中提取高清原图 URL。
* **稳定高级方案**：如果前端签名算法频繁更新，则在 Node.js 中使用 `Puppeteer`（无头浏览器）直接模拟用户打开链接，等待图片加载完成后拦截网络请求或解析 DOM。

### 核心解析逻辑（`parser.js`）

```javascript
const axios = require('axios');
const cheerio = require('cheerio');
const puppeteer = require('puppeteer');

// 通用配置，模拟手机浏览器
const axiosConfig = {
  headers: {
    'User-Agent': 'Mozilla/5.0 (Linux; Android 10; K) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/122.0.0.0 Mobile Safari/537.36'
  }
};

/**
 * 1. 微信公众号解析
 */
async function parseWeChat(url) {
  const { data } = await axios.get(url, axiosConfig);
  const $ = cheerio.load(data);
  const images = [];
  
  // 公众号图片的真实地址通常在 data-src 中
  $('#js_content img').each((i, el) => {
    const src = $(el).attr('data-src') || $(el).attr('src');
    if (src) images.push(src);
  });
  return images;
}

/**
 * 2. 小红书图文解析 (静态网页提取法)
 */
async function parseXHS(url) {
  const { data } = await axios.get(url, axiosConfig);
  const $ = cheerio.load(data);
  
  // 小红书的数据通常脱水在 window.__INITIAL_STATE__ 中
  const scriptContent = $('script').filter((i, el) => {
    return $(el).html().includes('__INITIAL_STATE__');
  }).html();

  if (scriptContent) {
    // 提取 json 字符串
    const jsonStr = scriptContent.match(/__INITIAL_STATE__\s*=\s*({.+})/)[1];
    const state = JSON.parse(jsonStr);
    
    // 根据小红书当前的 state 结构获取图片列表
    // 注意：平台结构可能微调，此处为通用逻辑，通常在 note.noteDetailMap[id].note.imageList 中
    const noteId = Object.keys(state.note.noteDetailMap)[0];
    const imageList = state.note.noteDetailMap[noteId].note.imageList || [];
    
    return imageList.map(img => img.urlDefault || img.url);
  }
  
  throw new Error('未找到小红书数据');
}

/**
 * 3. 抖音图文解析 (Puppeteer 降维打击稳定方案)
 * 抖音的反爬和签名极为严格，静态请求极易被拦截或返回验证码，用无头浏览器最稳妥
 */
async function parseDouyin(url) {
  const browser = await puppeteer.launch({ headless: true });
  const page = await browser.newPage();
  
  // 设置手机端 Viewport
  await page.setViewport({ width: 375, height: 812, isMobile: true });
  await page.setUserAgent(axiosConfig.headers['User-Agent']);
  
  await page.goto(url, { waitUntil: 'networkidle2' });
  
  // 等待渲染完成后，提取图片 DOM
  // 抖音图文通常在特定的 swiper 或 container 容器中
  const images = await page.evaluate(() => {
    // 这里的选择器需根据抖音实时网页结构微调，目前通用寻找 img
    // 也可以通过拦截网络请求获取 api 返回的 json
    const imgEls = document.querySelectorAll('img'); 
    return Array.from(imgEls)
      .map(img => img.src)
      .filter(src => src.includes('pstatp.com') || src.includes('byteimg.com')); // 过滤出抖音图文素材域名的图片
  });
  
  await browser.close();
  return images;
}

module.exports = { parseWeChat, parseXHS, parseDouyin };

```

### 服务端路由入口（`server.js`）

```javascript
const express = require('express');
const { parseWeChat, parseXHS, parseDouyin } = require('./parser');
const app = express();

app.use(express.json());

// 跨域处理（研发阶段）
app.use((req, res, next) => {
  res.header('Access-Control-Allow-Origin', '*');
  res.header('Access-Control-Allow-Headers', 'Content-Type');
  next();
});

app.post('/api/extract-images', async (req, res) => {
  const { url } = req.body;
  if (!url) return res.status(400).json({ error: '链接不能为空' });

  try {
    let images = [];
    if (url.includes('mp.weixin.qq.com')) {
      images = await parseWeChat(url);
    } else if (url.includes('xiaohongshu.com') || url.includes('xhslink.com')) {
      images = await parseXHS(url);
    } else if (url.includes('douyin.com')) {
      images = await parseDouyin(url);
    } else {
      return res.status(400).json({ error: '暂不支持该平台链接' });
    }
    
    res.json({ success: true, data: images });
  } catch (error) {
    res.status(500).json({ success: false, error: error.message });
  }
});

app.listen(3000, () => console.log('Server running on port 3000'));

```

## Vue3 前端代码实现

前端提供一个输入框和批量卡片展示区域。由于提取出来的图片直接放入 `<img>` 标签会触发各大平台的**防盗链机制**（返回 403 错误或不显示），我们需要在 HTML 中加入 `<meta name="referrer" content="no-referrer">` 来绕过防盗链。

```vue
<template>
  <div class="analyzer-container">
    <div class="input-section">
      <input 
        v-model="inputUrl" 
        placeholder="粘贴公众号、小红书或抖音图文链接..." 
        @keyup.enter="handleExtract"
      />
      <button :disabled="loading" @click="handleExtract">
        {{ loading ? '解析中...' : '提取图片' }}
      </button>
    </div>

    <!-- 批量图片物料展示画布/卡片布局 -->
    <div v-if="images.length" class="gallery-grid">
      <div v-for="(img, index) in images" :key="index" class="image-card">
        <!-- 这里的 referrerPolicy="no-referrer" 是破解防盗链的关键 -->
        <img :src="img" referrerPolicy="no-referrer" alt="extracted content" loading="lazy" />
        <div class="card-actions">
          <button @click="downloadImg(img, index)">下载</button>
        </div>
      </div>
    </div>
  </div>
</template>

<script setup>
import { ref } from 'vue';

const inputUrl = ref('');
const loading = ref(false);
const images = ref([]);

const handleExtract = async () => {
  if (!inputUrl.value) return;
  loading.value = true;
  images.value = [];
  
  try {
    const response = await fetch('http://localhost:3000/api/extract-images', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ url: inputUrl.value })
    });
    const result = await response.json();
    if (result.success) {
      images.value = result.data;
    } else {
      alert('解析失败: ' + result.error);
    }
  } catch (err) {
    alert('网络请求失败');
  } finally {
    loading.value = false;
  }
};

// 后端虽然绕过了防盗链，前端直接下载外链依然会受限，建议下载走 Node 层的图片转 Base64/Blob 接口
const downloadImg = (url, index) => {
  // 简易逻辑：直接开新窗口，生产环境建议让后端代理下载
  window.open(url, '_blank');
};
</script>

<style scoped>
```

## 避坑指南与高级调研技巧

> **关于防盗链（403 Forbidden）**
> 前端加入 `referrerPolicy="no-referrer"` 能解决 **90%** 的网页端查看问题。如果有些平台在某些浏览器下依然裂图，最佳方案是在 Node 后端做一个 **图片中转代理路由**：
> `GET /api/image-proxy?url=BASE64_IMAGE_URL`
> 后端接收到请求后，用 `axios.get(url, { responseType: 'stream', headers: { Referer: '' } })` 去抓取图片原数据，然后 `pipe(res)` 返回给前端，彻底抹掉来源信息。

> **关于小红书短链（xhslink.com）**
> 用户复制出来的链接经常是短链。Node.js 拿到短链后，需要先通过 `axios` 请求它（禁止重定向），获取 `response.headers.location` 拿到真实的**长链接**，再用长链接去匹配具体的解析逻辑。
